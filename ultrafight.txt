      updateWinnerSelect();

  function updatePlayerList() {
    playerListDisplay.innerHTML = '';
    players.forEach(p => {
      const li = document.createElement('li');
      li.textContent = p;
      playerListDisplay.appendChild(li);
    });
  }

  function updateWinnerSelect() {
    winnerSelect.innerHTML = '<option value="">Select winner...</option>';
    players.forEach(p => {
      const opt = document.createElement('option');
      opt.value = p;
      opt.textContent = p;
      winnerSelect.appendChild(opt);
    });
  }

  // Start game
  startBtn.addEventListener('click', () => {
    if (players.length < 2) {
      alert('Add at least 2 players to start.');
      return;
    }
    startupSection.style.display = 'none';
    gameplaySection.style.display = 'block';
    leaderboard = {};
    players.forEach(p => leaderboard[p] = 0);
    updateLeaderboard();
  });

    const weap = getRandomFromArray(WEAPONS);
    const twist = getRandomFromArray(TWISTS);

    let loc = null;
    if (useLocationsCheckbox.checked) {
      loc = getRandomFromArray(LOCATIONS);
    }
    let scen = null;
    if (useScenariosCheckbox.checked) {
      scen = getRandomFromArray(SCENARIOS);
    }

    // Create cards and add them with animation
    addCard('Character', char.name);
    addCard('Weapon', weap.name);
    addCard('Twist', twist.name);
    if (loc) addCard('Location', loc.name);
    if (scen) addCard('Scenario', scen.name);

    // Show fight context
    let context = `The battle starts! Your fighter is ${char.name}, armed with ${weap.name}. Twist: ${twist.name}.`;
    if (loc) context += ` Location: ${loc.name}.`;
    if (scen) context += ` Scenario: ${scen.name}.`;

    fightContext.textContent = context;

  function addCard(type, name) {
    const card = document.createElement('div');
    card.className = 'card';
    card.innerHTML = `<strong>${type}</strong><br>${name}`;
    cardsRow.appendChild(card);
    // Trigger flip animation manually
    setTimeout(() => {
      card.style.opacity = '1';
    }, 50);
  }

  // Declare winner and update leaderboard
  gmWinBtn.addEventListener('click', () => {
    const winner = winnerSelect.value;
    if (!winner) {
      alert('Select a winner first.');
      return;
    }
    leaderboard[winner]++;
    updateLeaderboard();
  });

  function updateLeaderboard() {
    leaderboardList.innerHTML = '';
    const sorted = Object.entries(leaderboard).sort((a, b) => b[1] - a[1]);
    sorted.forEach(([player, score]) => {
      const li = document.createElement('li');
      li.textContent = `${player}: ${score} points`;
      leaderboardList.appendChild(li);
    });
  }

  function getRandomFromArray(arr) {
    return arr[Math.floor(Math.random() * arr.length)];
  }

  // Load custom deck file
  deckUpload.addEventListener('change', (e) => {
    const file = e.target.files[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = (evt) => {
      try {
        const data = JSON.parse(evt.target.result);
        if (data.characters) CHARACTERS = data.characters;
        if (data.weapons) WEAPONS = data.weapons;
        if (data.twists) TWISTS = data.twists;
        if (data.locations) LOCATIONS = data.locations;
        if (data.scenarios) SCENARIOS = data.scenarios;
        alert('Custom deck loaded!');
      } catch {
        alert('Invalid JSON file.');
      }
    };
    reader.readAsText(file);
  });
